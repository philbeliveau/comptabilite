```python
@extension_endpoint("creer_facture", ["POST"])
def creer_facture(self):
    """POST: Create a new AR invoice from form data."""
    form = request.form

    # Parse client info
    nom_client = form.get("nom_client", "").strip()
    date_str = form.get("date", "")
    echeance_str = form.get("date_echeance", "")
    notes = form.get("notes", "")

    date_facture = datetime.date.fromisoformat(date_str)
    date_echeance = datetime.date.fromisoformat(echeance_str)

    # Parse dynamic line items (indexed: description_0, quantite_0, prix_unitaire_0, tps_0, tvq_0, ...)
    lignes = []
    i = 0
    while f"description_{i}" in form:
        desc = form.get(f"description_{i}", "").strip()
        qte = int(form.get(f"quantite_{i}", "1"))
        prix = Decimal(form.get(f"prix_unitaire_{i}", "0"))
        tps = f"tps_{i}" in form  # Checkbox: present = checked
        tvq = f"tvq_{i}" in form
        if desc and prix > 0:
            lignes.append(LigneFacture(
                description=desc,
                quantite=qte,
                prix_unitaire=prix,
                tps_applicable=tps,
                tvq_applicable=tvq,
            ))
        i += 1

    if not lignes or not nom_client:
        # Redirect back with error (simple approach: just redirect)
        return redirect(request.referrer or "/")

    # Create invoice
    registre = RegistreFactures()
    numero = registre.prochain_numero(date_facture.year)
    facture = Facture(
        numero=numero,
        nom_client=nom_client,
        date=date_facture,
        date_echeance=date_echeance,
        lignes=lignes,
        statut=InvoiceStatus.DRAFT,
        notes=notes,
    )
    registre.ajouter(facture)

    # Generate and append Beancount journal entry
    ecriture = generer_ecriture_facture(facture)
    journal_path = Path("ledger/factures/journal.beancount")
    journal_path.parent.mkdir(parents=True, exist_ok=True)
    with open(journal_path, "a", encoding="utf-8") as f:
        f.write("\n" + ecriture + "\n")

    return redirect(request.referrer or "/")
```

```javascript
var arLineCount = 1;

function addARLine() {
  var idx = arLineCount++;
  var container = document.getElementById('ar-lines');
  var div = document.createElement('div');
  div.className = 'cqc-form-line';
  div.setAttribute('data-index', idx);
  div.innerHTML = '<input type="text" name="description_' + idx + '" placeholder="Description" required style="flex:3;">'
    + '<input type="number" name="quantite_' + idx + '" value="1" min="1" style="width:60px;">'
    + '<input type="number" name="prix_unitaire_' + idx + '" step="0.01" placeholder="Prix/u" required style="flex:1;">'
    + '<label><input type="checkbox" name="tps_' + idx + '" checked> TPS</label>'
    + '<label><input type="checkbox" name="tvq_' + idx + '" checked> TVQ</label>';
  container.appendChild(div);
  div.querySelectorAll('input').forEach(function(inp) {
    inp.addEventListener('input', updateARTotals);
    inp.addEventListener('change', updateARTotals);
  });
}

function updateARTotals() {
  var subtotal = 0, tpsBase = 0, tvqBase = 0;
  document.querySelectorAll('#ar-lines .cqc-form-line').forEach(function(line) {
    var idx = line.getAttribute('data-index');
    var qte = parseFloat(line.querySelector('[name="quantite_' + idx + '"]').value) || 0;
    var prix = parseFloat(line.querySelector('[name="prix_unitaire_' + idx + '"]').value) || 0;
    var lineTotal = qte * prix;
    subtotal += lineTotal;
    if (line.querySelector('[name="tps_' + idx + '"]').checked) tpsBase += lineTotal;
    if (line.querySelector('[name="tvq_' + idx + '"]').checked) tvqBase += lineTotal;
  });
  var tps = tpsBase * 0.05;
  var tvq = tvqBase * 0.09975;
  document.getElementById('ar-subtotal').textContent = subtotal.toFixed(2);
  document.getElementById('ar-tps').textContent = tps.toFixed(2);
  document.getElementById('ar-tvq').textContent = tvq.toFixed(2);
  document.getElementById('ar-total').textContent = (subtotal + tps + tvq).toFixed(2);
}

// Attach listeners to initial line
document.querySelectorAll('#ar-lines input').forEach(function(inp) {
  inp.addEventListener('input', updateARTotals);
  inp.addEventListener('change', updateARTotals);
});

function toggleForm(id) {
  var el = document.getElementById(id);
  el.style.display = el.style.display === 'none' ? '' : 'none';
}
```

```python
from compteqc.fournisseurs.modeles import FactureFournisseur, LigneFactureFournisseur, BillStatus
from compteqc.fournisseurs.journal import generer_ecriture_facture_fournisseur
from compteqc.fournisseurs.registre import RegistreFournisseurs

@extension_endpoint("creer_facture_fournisseur", ["POST"])
def creer_facture_fournisseur(self):
    """POST: Create a new AP bill from form data."""
    form = request.form

    fournisseur = form.get("fournisseur", "").strip()
    numero_reference = form.get("numero_reference", "").strip()
    date_facture_str = form.get("date_facture", "")
    date_echeance_str = form.get("date_echeance", "")
    notes = form.get("notes", "")

    date_facture = datetime.date.fromisoformat(date_facture_str)
    date_echeance = datetime.date.fromisoformat(date_echeance_str)

    # Parse dynamic line items
    # Fields: ap_description_0, ap_montant_0, ap_categorie_0, ap_tps_0, ap_tvq_0, ap_taux_itc_0, ap_taux_itr_0
    lignes = []
    i = 0
    while f"ap_description_{i}" in form:
        desc = form.get(f"ap_description_{i}", "").strip()
        montant = Decimal(form.get(f"ap_montant_{i}", "0"))
        categorie = form.get(f"ap_categorie_{i}", "Depenses:Divers")
        tps = f"ap_tps_{i}" in form
        tvq = f"ap_tvq_{i}" in form
        taux_itc = Decimal(form.get(f"ap_taux_itc_{i}", "1.0"))
        taux_itr = Decimal(form.get(f"ap_taux_itr_{i}", "1.0"))
        if desc and montant > 0:
            lignes.append(LigneFactureFournisseur(
                description=desc,
                montant=montant,
                categorie_depense=categorie,
                tps_applicable=tps,
                tvq_applicable=tvq,
                taux_itc=taux_itc,
                taux_itr=taux_itr,
            ))
        i += 1

    if not lignes or not fournisseur:
        return redirect(request.referrer or "/")

    registre = RegistreFournisseurs()
    numero_interne = registre.prochain_numero(date_facture.year)
    bill = FactureFournisseur(
        numero_reference=numero_reference,
        numero_interne=numero_interne,
        fournisseur=fournisseur,
        date_facture=date_facture,
        date_echeance=date_echeance,
        lignes=lignes,
        statut=BillStatus.RECEIVED,
        notes=notes,
    )
    registre.ajouter(bill)

    # Generate and append Beancount journal entry
    ecriture = generer_ecriture_facture_fournisseur(bill)
    journal_path = Path("ledger/fournisseurs/journal.beancount")
    journal_path.parent.mkdir(parents=True, exist_ok=True)
    with open(journal_path, "a", encoding="utf-8") as f:
        f.write("\n" + ecriture + "\n")

    return redirect(request.referrer or "/")
```

```javascript
    var montant = parseFloat(line.querySelector('[name="ap_montant_' + idx + '"]').value) || 0;
    subtotal += montant;
    var tpsOn = line.querySelector('[name="ap_tps_' + idx + '"]').checked;
    var tvqOn = line.querySelector('[name="ap_tvq_' + idx + '"]').checked;
    var tauxItc = parseFloat(line.querySelector('[name="ap_taux_itc_' + idx + '"]').value) || 0;
    var tauxItr = parseFloat(line.querySelector('[name="ap_taux_itr_' + idx + '"]').value) || 0;
    var lineTps = tpsOn ? montant * 0.05 : 0;
    var lineTvq = tvqOn ? montant * 0.09975 : 0;
    tpsTotal += lineTps;
    tvqTotal += lineTvq;
    itcTotal += lineTps * tauxItc;
    itrTotal += lineTvq * tauxItr;
  });
  document.getElementById('ap-subtotal').textContent = subtotal.toFixed(2);
  document.getElementById('ap-tps').textContent = tpsTotal.toFixed(2);
  document.getElementById('ap-tvq').textContent = tvqTotal.toFixed(2);
  document.getElementById('ap-total').textContent = (subtotal + tpsTotal + tvqTotal).toFixed(2);
  document.getElementById('ap-itc').textContent = itcTotal.toFixed(2);
  document.getElementById('ap-itr').textContent = itrTotal.toFixed(2);
}
```
